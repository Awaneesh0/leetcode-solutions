class Solution {
    public boolean isPowerOfFour(int n) {
        // 1. Must be positive
        // 2. Must be a power of two (only one 1-bit)
        // 3. That 1-bit must land on an even index (matched against 0x55555555)
        return (n > 0) && ((n & (n - 1)) == 0) && ((n & 0x55555555) != 0);
    }
}
